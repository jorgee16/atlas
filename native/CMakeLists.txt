cmake_minimum_required(VERSION 3.22)
project(atlas_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(atlas_core STATIC src/binary.cpp src/routing.cpp src/tolls.cpp src/toll_crossings.cpp src/search.cpp src/guidance.cpp src/motion.cpp)
set_target_properties(atlas_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(atlas_core PUBLIC include)
target_compile_options(atlas_core PRIVATE -Wall -Wextra -Wpedantic)
if(ANDROID)
  add_library(atlas_jni SHARED src/jni.cpp)
  target_link_libraries(atlas_jni PRIVATE atlas_core)
else()
  enable_testing()
  add_executable(atlas_core_tests tests/core_tests.cpp)
  target_link_libraries(atlas_core_tests PRIVATE atlas_core)
  add_test(NAME atlas_core_tests COMMAND atlas_core_tests)
  add_executable(atlas_navigation_replay_tests tests/navigation_replay_tests.cpp)
  target_link_libraries(atlas_navigation_replay_tests PRIVATE atlas_core)
  add_test(NAME atlas_navigation_replay_tests COMMAND atlas_navigation_replay_tests)
  add_executable(atlas_toll_crossing_tests tests/toll_crossing_tests.cpp)
  target_link_libraries(atlas_toll_crossing_tests PRIVATE atlas_core)
  add_test(NAME atlas_toll_crossing_tests COMMAND atlas_toll_crossing_tests)
  add_executable(atlas_toll_tests tests/toll_tests.cpp)
  target_link_libraries(atlas_toll_tests PRIVATE atlas_core)
  add_test(NAME atlas_toll_tests COMMAND atlas_toll_tests ${CMAKE_CURRENT_SOURCE_DIR}/../android/native-app/src/main/assets/tolls/Portugal.csv ${CMAKE_CURRENT_SOURCE_DIR}/../android/native-app/src/main/assets/toll-charges/Portugal.csv ${CMAKE_CURRENT_SOURCE_DIR}/../android/native-app/src/main/assets/toll-fares/Portugal.csv)
  add_executable(atlas_search src/search_cli.cpp)
  target_link_libraries(atlas_search PRIVATE atlas_core)
  add_executable(atlas_toll_quote src/toll_quote_cli.cpp)
  target_link_libraries(atlas_toll_quote PRIVATE atlas_core)
  add_executable(atlas_route src/cli.cpp)
  target_link_libraries(atlas_route PRIVATE atlas_core)
endif()
